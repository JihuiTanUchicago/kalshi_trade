        \begin{itemize}

        \end{itemize}

% SLIDE 11: CONCLUSION
% =========================
\renewcommand{\sectiontitle}{Conclusion}
\section{\sectiontitle}

\begin{frame}{\sectiontitle}
\footnotesize
\textbf{What we did}
\begin{itemize}
    \itemj Built an end-to-end pipeline: Kalshi + Coinbase data, 20s$\to$1-min features, MySQL, shared research and live trading engine
    \itemj Re-ran the benchmark and Models 1--5 on live data from Mar 7 02:11 through Mar 12 06:04 UTC, then regenerated every slide plot from that refresh
    \itemj Expiry-model threshold tuning now requires at least 10 validation trades when possible, with fallback to the unconstrained search if no rule meets that floor
\end{itemize}

\vspace{0.1cm}
\textbf{Results (in-sample vs out-of-sample)}
\begin{itemize}
    \itemj Linear models stayed negative OOS: benchmark \textbf{-\$18.26} and L1 logistic \textbf{-\$54.49}, despite modest ranking lift (AUC $\sim$0.66)
    \itemj Expiry-resolution tree models kept strong ranking quality (test AP/AUC roughly 0.82--0.88), but realized tradable edge was sparse on the refreshed OOS slice
    \itemj GBM finished slightly positive at \textbf{+\$0.13} on 1 OOS trade; RF lost \textbf{-\$0.68} on 4 trades; Bagging traded \textbf{11} times for \textbf{-\$0.35}; AdaBoost lost \textbf{-\$0.23} on 8 trades
\end{itemize}

\vspace{0.1cm}
\textbf{Takeaways}
\begin{itemize}
    \itemj Predictive ranking alone is not enough; threshold selection and execution assumptions dominate realized PnL once we demand live, non-synthetic trades
    \itemj More data, more robust walk-forward validation, and transaction-cost-aware tuning are still required before promoting any tree model into production
\end{itemize}
\end{frame}

% SLIDE 12: NEXT STEPS

\renewcommand{\sectiontitle}{Next Steps}
\section{\sectiontitle}

\begin{frame}{\sectiontitle}
\footnotesize
\textbf{Models}
\begin{itemize}
    \itemj Expand to longer walk-forward windows and retune thresholds with a minimum-trade, transaction-cost-aware objective so inactivity does not dominate model selection
    \itemj Use feature importance (e.g., Shapley) to shrink the 39-feature tree set and test whether we can preserve AP/AUC while improving trade density
\end{itemize}

\vspace{0.1cm}
\textbf{Execution \& risk}
\begin{itemize}
    \itemj Harden live engine: slippage and partial fills, expiry-boundary rules, quote-age filters, and drawdown / position limits
\end{itemize}

\vspace{0.1cm}
\textbf{Monitoring}
\begin{itemize}
    \itemj Add calibration drift, threshold stability, and per-model trade-density panels so we can see when strong AP/AUC is failing to convert into executable edge
\end{itemize}
\end{frame}
